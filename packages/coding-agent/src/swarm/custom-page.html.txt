<p id="status" role="status">Connecting…</p>
<iframe id="page" title="Narrator custom page" sandbox="allow-scripts" referrerpolicy="no-referrer" hidden></iframe>
</main>
<script nonce="{{NONCE}}">
(() => {
	"use strict";
	const config = {{CONFIG}};
	const frame = document.getElementById("page");
	const status = document.getElementById("status");
	const navigation = document.getElementById("navigation");
	const target = document.getElementById("target");
	const open = document.getElementById("open");
	const reducedMotion = matchMedia("(prefers-reduced-motion: reduce)");
	document.documentElement.dataset.ompReducedMotion = config.reducedMotion ? "true" : "false";
	let state;
	let revision;
	let pending;
	let busy = false;
	let enforced = false;
	function showStatus(message) {
		status.textContent = message;
		status.hidden = !message;
	}
	if (!window.trustedTypes || typeof window.Sanitizer !== "function" || typeof Element.prototype.setHTML !== "function") {
		showStatus("Custom pages require a browser with Trusted Types and the native HTML Sanitizer API.");
		return;
	}
	function sendData() {
		if (state?.view && frame.contentWindow) frame.contentWindow.postMessage({ type: "omp:swarm:data", view: state.view, reducedMotion: config.reducedMotion || reducedMotion.matches }, "*");
	}
	reducedMotion.addEventListener("change", sendData);
	window.addEventListener("message", event => {
		if (!event.isTrusted || event.source !== frame.contentWindow || event.origin !== "null") return;
		const message = event.data;
		if (!message || typeof message !== "object" || Array.isArray(message)) return;
		if (message.type === "omp:swarm:ready" && message.enforced === true) {
			enforced = true;
			frame.hidden = false;
			sendData();
			return;
		}
		if (message.type === "omp:swarm:error" && typeof message.message === "string") {
			showStatus(`Custom page: ${message.message.slice(0, 280)}`);
			return;
		}
		if (message.type !== "omp:swarm:navigate" || !enforced || busy) return;
		const action = message.action;
		if (!action || typeof action !== "object" || Array.isArray(action)) return;
		if (!["open_conv", "board_post", "url", "file"].includes(action.type)) return;
		if (action.type === "board_post") {
			if (!Number.isSafeInteger(action.target) || action.target < 1) return;
		} else if (typeof action.target !== "string" || !action.target || action.target.length > 4096) return;
		pending = { type: action.type, target: action.target };
		const labels = { open_conv: "Conversation", board_post: "Board post", url: "HTTPS link", file: "Artifact" };
		target.textContent = `${labels[action.type]}: ${action.type === "board_post" ? "#" : ""}${action.target}`;
		navigation.hidden = false;
	});
	document.getElementById("cancel").addEventListener("click", () => {
		pending = undefined;
		navigation.hidden = true;
	});
	open.addEventListener("click", async event => {
		if (!event.isTrusted || !pending || busy) return;
		const action = pending;
		busy = true;
		open.disabled = true;
		try {
			const response = await fetch(config.navigate, {
				method: "POST",
				headers: { "Content-Type": "application/json", "X-OMP-Swarm": "navigate" },
				body: JSON.stringify(action),
			});
			if (!response.ok) throw new Error(await response.text());
			pending = undefined;
			navigation.hidden = true;
			showStatus(state?.error ?? "");
		} catch (error) { showStatus(String(error)); }
		finally { busy = false; open.disabled = false; }
	});
	const events = new EventSource(config.events);
	events.addEventListener("message", event => {
		try { state = JSON.parse(event.data); }
		catch { showStatus("The board update could not be read."); return; }
		document.getElementById("title").textContent = state.title;
		document.getElementById("sequence").textContent = `#${state.head}`;
		document.title = `${state.title} · Swarm`;
		if (state.revision !== revision) {
			revision = state.revision;
			pending = undefined;
			navigation.hidden = true;
			enforced = false;
			frame.hidden = true;
			if (revision) frame.src = `${config.frame}?revision=${encodeURIComponent(revision)}`;
			else frame.removeAttribute("src");
		} else sendData();
		showStatus(state.error ?? (revision ? "" : "Waiting for view.html."));
	});
	events.addEventListener("error", () => showStatus("Connection interrupted. Reconnecting…"));
	window.addEventListener("pagehide", () => events.close(), { once: true });
})();
</script>
